	metweb.common.Date2struct=function(date) {
	  var yr = date.getUTCFullYear()+'';
		var mo = (date.getUTCMonth()+1)+'';
		if (mo.length == 1) {
			mo = "0"+mo;
		}
		var da = date.getUTCDate()+'';
		if (da.length == 1) {
			da = "0"+da;
		}
		var hr = date.getUTCHours()+'';
		if (hr.length == 1) {
			hr = "0"+hr;
		}
		var mi = date.getUTCMinutes()+'';
		if (mi.length == 1) {
			mi = "0"+mi;
		}
		var se = date.getUTCSeconds()+'';
		if (se.length == 1) {
			se = "0"+se;
		}
		var ms = date.getUTCMilliseconds()+'';
		if (ms.length == 1) {
			ms = "00"+ms;
		} else if (ms.length == 2) {
			ms = "0"+ms;
		}
		
		return {'yr':yr, 'mo':mo, 'da':da, 'hr':hr, 'mi':mi, 'se':se, 'ms':ms};
	};

	// handles various iso8601 combinations and returns an epoch in JS milliseconds
	metweb.common.iso2epoch=function(iso){
		if (!iso) { 
			return 0; 
		}
		// first, let the javascript Date library have a crack at it
		// Known: Date doesn't handle YYYY. 
		// Otherwise, most requests will fall into here. And Date does a very good consistent job
		var epoch;
		if (iso.length > 4) {
			var date = new Date(iso);
			if (date) {
				epoch = date.getTime();
				if (epoch != undefined) {
					return epoch;
				}
			}
			metweb.common.dumper("iso2epoch and Date may have had trouble to convert "+iso+" into an epoch");
		} else if (iso.length == 4) {
			var s = {
				'yr':iso,
				'mo':'01',
				'da':'01',
				'hr':'00',
				'mi':'00',
				'se':'00',
				'ms':'000',
			};
			epoch = metweb.common.Date2epoch(metweb.common.struct2Date(s));
			return epoch;
		} else {
			metweb.common.dumper("iso2epoch and Date may have had trouble to convert "+iso+" into an epoch");
		}
		return 0;
	}

	// figure out the time string format from an iso time string
	metweb.common.iso2fmt=function(iso){
		if (!iso) { 
			return ''; 
		}
		var fmt = iso;
		fmt = fmt.replace(/^\d{4}-\d{2}-\d{2}/,'%F');
		fmt = fmt.replace(/^\d{4}-\d{2}/,'%Y-%m');
		fmt = fmt.replace(/^\d{4}/,'%Y');
		fmt = fmt.replace(/\d{2}:\d{2}:\d{2}/,'%T');
		fmt = fmt.replace(/\d{2}:\d{2}/,'%H:%M');
		fmt = fmt.replace(/\.\d{3}/,'.%f');
		return fmt;
	}

	// js doesn't have posix strftime. This is an 'our' strftime, but in UTC(Z) only.
	metweb.common.strftime=function(fmt,e){
		var s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		var tstr = fmt;
		tstr = tstr.replace('%F',[s.yr,s.mo,s.da].join('-'));
		tstr = tstr.replace('%Y',s.yr);
		tstr = tstr.replace('%m',s.mo);
		tstr = tstr.replace('%d',s.da);
		tstr = tstr.replace('%T',[s.hr,s.mi,s.se].join(':'));
		tstr = tstr.replace('%H',s.hr);
		tstr = tstr.replace('%M',s.mi);
		tstr = tstr.replace('%S',s.se);
		tstr = tstr.replace('%f',s.ms);
		return tstr;
	}

	// Get start of climatological month
	metweb.common.epoch_start_climate_month = function (e) {
		var s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		//if the date is already the first and before 06:01z, set to previous month first
		if (parseInt(s.da, 10) == 1 && (parseInt(s.hr, 10) == 6 && parseInt(s.mi, 10) < 1)){
			e -= 24 * 60 * 60 * 1000;
			s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		}

		//set day to 1st of current month
		while (parseInt(s.da, 10) > 1) {
			e -= 24 * 60 * 60 * 1000;
			s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		}

		s.hr = "06";
		s.mi = "01";
		s.se = "00";
		s.ms = "000";
		return metweb.common.struct2epoch(s);
	}

	// Get end of climatological month
	metweb.common.epoch_end_climate_month = function (e) {
		var s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		if (parseInt(s.da, 10) > 1 || (parseInt(s.hr, 10) >= 6 && parseInt(s.mi, 10) >= 1)){
			e = metweb.common.epoch_plus_month(e);
			s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		}

		s.hr = "06";
		s.mi = "00";
		s.se = "00";
		s.ms = "000";
		return metweb.common.struct2epoch(s);
	}

	metweb.common.epoch_minus_month = function (e) {
		var s = metweb.common.Date2struct(metweb.common.epoch2Date(e));
		var s2 = s;
		var e2 = e;
		while (s2.mo == s.mo) {
			e2 -= 24 * 60 * 60 * 1000;
			s2 = metweb.common.Date2struct(metweb.common.epoch2Date(e2));
		}
		return e2;
	}
	metweb.common.epoch_plus_month=function(e) {

	}

	metweb.common.asktime=function(dt){
		var len = dt.length;
		var now = metweb.common.time();
		var ns = metweb.common.Date2struct(metweb.common.epoch2Date(now));
		switch(len){
			case 17: //YYYYMMDDHHMMSSsss
			case 14: //YYYYMMDDHHMMSS 
			case 12: //YYYYMMDDHHMM
				var s = {
						'yr':dt.substring(0,4),
						'mo':dt.substring(4,6),
						'da':dt.substring(6,8),
						'hr':dt.substring(8,10),
						'mi':dt.substring(10,12),
						'se':'00',
						'ms':'000'
					};
				return metweb.common.Date2epoch(metweb.common.struct2Date(s));

			case 8:	// YYMMDDHH
				var s = {
						'yr':'20'+dt.substring(0,2),
						'mo':dt.substring(2,4),
						'da':dt.substring(4,6),
						'hr':dt.substring(6),
						'mi':ns.mi,
						'se':'00',
						'ms':'000'
					};
				return metweb.common.Date2epoch(metweb.common.struct2Date(s));

			case 6:	// YYMMDD
				var s = {
						'yr':'20'+dt.substring(0,2),
						'mo':dt.substring(2,4),
						'da':dt.substring(4),
						'hr':ns.hr,
						'mi':ns.mi,
						'se':'00',
						'ms':'000'
					};
				return metweb.common.Date2epoch(metweb.common.struct2Date(s));

			case 4: // HHMM
				var s = {
						'yr':ns.yr,
						'mo':ns.mo,
						'da':ns.da,
						'hr':dt.substring(0,2),
						'mi':dt.substring(2),
						'se':'00',
						'ms':'000'
					};
				return metweb.common.Date2epoch(metweb.common.struct2Date(s));

			case 2:	// HH
				var s = {
						'yr':ns.yr,
						'mo':ns.mo,
						'da':ns.da,
						'hr':dt,
						'mi':'00',
						'se':'00',
						'ms':'000'
					};
				return metweb.common.Date2epoch(metweb.common.struct2Date(s));

			case 1:	// 0
				if( dt == "0" ){
					return now;
				}
				break;

			case 0: // empty
				return now;

			default:
				break;
		}

	};

	metweb.common.format_seconds_pretty=function(ep){
		// note expects epoch in seconds, not milliseconds.
		// Try to replicate perl's Time::Seconds->new(e)->pretty();
		var MIN = 60;
		var HR = 60*MIN;
		var DAY = 24*HR;
		//var MO = 30*DAY;
		var YR = 365*DAY;

		var t = parseInt(ep,10);
		var days = Math.floor(t/DAY);
		t -= days*DAY;
		var hours = Math.floor(t/HR);
		t -= hours*HR;
		var minutes = Math.floor(t/MIN);
		t -= minutes*MIN;
		var seconds = t;

		var str = '';
		if (days) {
			var dstr = (days<2)?"day":"days";
			str = str.concat(days, " ", dstr);
		}
		if (hours) {
			var hstr = (hours<2)?"hour":"hours";
			if (str.length) { str = str.concat(", "); }
			str = str.concat(hours, " ", hstr);
		}
		if (minutes) {
			var mstr = (minutes<2)?"minute":"minutes";
			if (str.length) { str = str.concat(", "); }
			str = str.concat(minutes, " ", mstr);
		}
		if (seconds) {
			var sstr = (seconds<2)?"second":"seconds";
			if (str.length) { str = str.concat(", "); }
			str = str.concat(seconds, " ", sstr);
		}
		return str;
	};

	metweb.common.grep=function(items,fn) {
		// http://gabrieleromanato.name/javascript-implementing-the-grep-function/
		// dojo._base.array.filter would be better, but...
		var filtered = [], len = items.length, i = 0;
		for (i; i < len; i++) {
			var item = items[i];
			var cond = fn(item);
			if (cond) {
				filtered.push(item);
			}
		}
		return filtered;
	};

	/* round to nearest. accepts strings or int/float */
	metweb.common.nearest=function(near, value) {
		if (value == null || near == null) {return NaN;}
		//if (typeof value === "object" || typeof near === "object") {return NaN;}
		if (isNaN(value) || isNaN(near)) {return NaN;}
		var value_f = (typeof value === "string")?parseFloat(value):value;
		var near_f = (typeof near === "string")?parseFloat(near):near;
		var answer = ( near_f * Math.round(value_f / near_f) );
		// javascript has a known float bug, use toFixed
		return parseFloat(answer.toFixed(13));
		//return (0.0000000000001 * Math.round(answer/0.0000000000001));
	};

	metweb.common.ourXhrGet=function(_o1){
		var path = (_o1.path == null) ? '' : _o1.path;
		/* according to dojo, xhrGet only accepts parameters in one possible field
 			* 'content': k/v hash, but seems to work with array ref if k,v pairs.
 			* !never values only!
 			* k v pairs converted noto url as &k=v in the request
 			*
 			* For backwards compabibility if ourXhrGet is passed 'params' it will
 			* get passed into 'content', until all code that uses Get gets fixed.
 			*
 			* Therefore when using Get the JSON handler must always expect a
 			* hash in 'params'.
 			*/

		var default_headers = { "Content-Type": "application/json"};
		dojo.xhrGet({
			url: "/json/" + path + _o1.method,
			/* 'content' preferred. 'params' for backcompat. MUST be k,v*/
			content: _o1.content||_o1.params, 
			headers: default_headers,
			/* FIXME when this is ready: http://cvs.isb.ec.gc.ca/tktview?tn=2234 
			*headers: _o1.headers?{...default_headers, ..._o1.headers}:default_headers,
			*/
			handleAs: _o1.handleAs||"json",
			sync: _o1.sync||false,
			error: function(error){
				/* alert the error */
				if (error.name){
					metweb.common.error(
						_o1.path+_o1.method+": error\nname="+error.name+
						"\nparams="+window.JSON.stringify( _o1.content||_o1.params,null," ")+
						"\nmessage="+error.message+
						"\nfileName="+(error.fileName||'')+
						"\nlineNumber="+(error.lineNumber||'')+
						"\nstack="+(error.stack||'')
					);
				}else{
					metweb.common.error(_o1.method+": error="+error);
				}
				/* Give application an opportunity to handle the error or recover */
				if (_o1.error){
					_o1.error(error);
				}
			},
			load: function(data, a){
				if (data.error) {
					if (data.error.name) {
						metweb.common.error(_o1.method+": data.error\nname="+data.error.name+
							"\nmessage="+data.error.message
						);
					} else {
						metweb.common.error(_o1.method+": data.error="+data.error);
					}
				} else if (typeof data.result !== 'undefined' || _o1.handleAs == 'text') { // yes the !== is intentional
					if (_o1.load) {
						_o1.load(data, a);
					}
				} else {
					metweb.common.error(_o1.method+": unknown data object");
				}
			}
		});
	};

	metweb.common.ourXhrPost=function(_o1) {
		var path = (_o1.path == null) ? '' : _o1.path;
		/* according to dojo, xhrPost will exclusively accept either
 			* 'postData.params': (preferred) as array ref of vals but hash is ok
 			*      OR
 			* 'content': k/v hash, but Post drops the keys and 
 			*  converts to array ref of values only
 			*
 			* ourXhrPost will NOT use xhrPost 'content'. It will assign 'content' hash
 			* to 'params' to preserve the hash, and the handler may expect the orig 
 			* hash set by the caller.
 			*
 			* Therefore when using Post the JSON handler must always expect an
 			* array values in 'params'. By definition they're values, unless was stuffed k,v.
 			*/
		var postDataHash = {version:"1.1",method:_o1.method,params:_o1.params||_o1.content};
		dojo.xhrPost(
			{
			url: "/json/"+ path +_o1.method,
			postData: dojo.toJson(postDataHash),
			headers: { "Content-Type": "application/json"},
			handleAs: "json",
			sync:_o1.sync||false,
			error: function(error){
				/* alert the error */
				if (error.name) {
					metweb.common.error(
						_o1.path+_o1.method+": error\nname="+error.name+
						"\nparams="+window.JSON.stringify(_o1.params,null," ")+
						"\nmessage="+error.message+
						"\nfileName="+(error.fileName||'')+
						"\nlineNumber="+(error.lineNumber||'')+
						"\nstack="+(error.stack||'')
					);
				} else {
					metweb.common.error(_o1.method+": error="+error);
				}
				/* give application an opportunity to handle the error or recover */
				if (_o1.error) {
					_o1.error(error);
				} 
			},
			load: function(data,a){
				if (data.error) {
					if (data.error.name) {
						metweb.common.error(_o1.method+": data.error\nname="+data.error.name+
							"\nmessage="+data.error.message
						);
					} else {
						metweb.common.error(_o1.method+": data.error="+data.error);
					}
				} else if (typeof data.result !== 'undefined') { // yes the !== is intentional
					if (_o1.load) {
						_o1.load(data, a);
					}
				} else {
					metweb.common.error(_o1.method+": unknown data object");
				}
			}
			} /* xhrPost parameters */
		); /* xhrPost */
	};

	metweb.common.xhr=function(_o1) {
		var p = (_o1.path === undefined) ? '' : _o1.path;

		var data;
		var url = "/json/"+p+_o1.method;
		if (_o1.type === 'GET') {
			var s = [];
			for (const key of Object.keys(_o1.params)) {
				s[s.length] = encodeURIComponent(key)+"="+encodeURIComponent(_o1.params[key]);
			}
			data = s.join("&");
		} else if (_o1.type === 'POST') {
			var body = {
				version:"1.1",
				method:_o1.method,
				params:_o1.params
			};
			data = JSON.stringify(body);
			var msize = metweb.common.max_request_size();
			if (data.length >= msize) {
				metweb.common.error(
					"Problem with request\n"+
					"url: "+p+_o1.method+"\n"+
					"Request size of "+data.length+
					" exceeds maximum of "+ msize
				);
				return;
			}
			if (_o1.urlextra) {
				url += "?"+_o1.urlextra;
			}
		}

		$.ajax({
			type:_o1.type,
			url:url,
			data: data,
			success:function (data, status) {
				if (status === 'success') {
					if (data.error) {
						metweb.common.error(
							"url: "+p+_o1.method+"\n"+
							"message: "+data.error.message
						);
						if (_o1.error) {
							_o1.error(data, status);
						}
					} else {
						_o1.load(data);
					}
				} else {
					metweb.common.error(
						"url: "+p+_o1.method+"\n"+
						"status: "+status
					);
					if (_o1.error) {
						_o1.error(data, status);
					}
				}
			},
			error:function (data, status) {
				metweb.common.error(
					"url: "+p+_o1.method+"\n"+
					"status: "+data.statusText
				);
				if (_o1.error) {
					_o1.error(data, status);
				}
			},
			contentType: "application/json",
			dataType: "json"
		});
	};

	metweb.common.set_and_propagate=function(srcObj,value) {
		if (!srcObj) {return;}
		if (srcObj.id.match("type_version")) { 
			// we should not copy/overwrite version information from an ob retrieve
			// the form version is immutable
			return; 
		}
		if (srcObj.type == 'checkbox') {
			if (value == 'on' || value == 'true' || value == true) {
			  srcObj.checked = true;
			} else {
			  srcObj.checked = false;
			}
		} else if (srcObj.value != value) {
			srcObj.value = value;
			if (srcObj.dispatchEvent) {
				var event = document.createEvent("HTMLEvents");
				event.initEvent("change", false, true);
				srcObj.dispatchEvent(event);
			} else if (srcObj.fireEvent) {
				srcObj.fireEvent("onchange");
			}
		}
		metweb.common.propagate(srcObj);
	};
